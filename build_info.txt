### 2. Mobile Agent实现（作为A2A客户端连接PC）

```kotlin
class MobileOpenClawAgent(
    private val pcServerUrl: String,
    private val platformService: PlatformService
) {
    private lateinit var a2aClient: A2AClient
    
    suspend fun initialize() {
        val transport = HttpJSONRPCClientTransport(url = pcServerUrl)
        val agentCardResolver = UrlAgentCardResolver(
            baseUrl = pcServerUrl.substringBeforeLast("/"),
            path = "/agent-card.json"
        )
        a2aClient = A2AClient(transport = transport, agentCardResolver = agentCardResolver)
        a2aClient.connect()
    }
    
    fun createAgent() = AIAgent(
        promptExecutor = simpleOpenAIExecutor(System.getenv("OPENAI_API_KEY")),
        llmModel = OpenAIModels.Chat.GPT4o,
        toolRegistry = ToolRegistry {
            tool(AppOperationTool(platformService))
            tool(FileOperationTool(platformService))
            tool(CommunicateWithPCTool(a2aClient))
        },
        strategy = strategy<String, String>("mobile-agent") {
            val nodeAnalyze by node<String, TaskType> { input ->
                withA2AAgentClient("pc-openclaw") {
                    sendTaskUpdate("Analyzing task: $input", TaskState.Working)
                }
                analyzeTaskType(input)
            }
            
            val nodeLocalExecute by node<TaskType, String> { taskType ->
                when (taskType) {
                    TaskType.LOCAL_APP -> executeLocalAppTask()
                    TaskType.PC_COORDINATION -> executeWithPCCoordination()
                    TaskType.PC_DELEGATE -> delegateToPC()
                }
            }
            
            val nodePCCommunicate by node<String, String> { message ->
                withA2AAgentClient("pc-openclaw") {
                    val response = sendMessageToAgent(
                        agentId = "pc-openclaw",
                        message = message
                    )
                    (response.data as Message).toKoogMessage().content
                }
            }
            
            edge(nodeStart forwardTo nodeAnalyze)
            edge(nodeAnalyze forwardTo nodeLocalExecute onCondition { it != TaskType.PC_DELEGATE })
            edge(nodeAnalyze forwardTo nodePCCommunicate onCondition { it == TaskType.PC_DELEGATE })
            edge(nodeLocalExecute forwardTo nodeFinish)
            edge(nodePCCommunicate forwardTo nodeFinish)
        }
    ) {
        install(A2AAgentClient) {
            this.a2aClients = mapOf("pc-openclaw" to a2aClient)
        }
    }
}
```

### 3. PC-Mobile通信工具

```kotlin
class CommunicateWithPCTool(
    private val a2aClient: A2AClient
) : Tool<CommunicateWithPCTool.Args, CommunicateWithPCTool.Result> {
    
    data class Args(
        val action: PCAction,
        val payload: String
    )
    
    sealed class PCAction {
        object QueryStatus : PCAction()
        object RequestResource : PCAction()
        object SyncData : PCAction()
        object ExecuteOnPC : PCAction()
    }
    
    data class Result(
        val success: Boolean,
        val response: String?,
        val error: String? = null
    )
    
    override val name = "communicate_with_pc"
    override val description = "与PC端OpenClaw进行通信，支持状态查询、资源请求、数据同步等"
    
    override suspend fun execute(args: Args): Result {
        return try {
            val message = Message(
                messageId = Uuid.random().toString(),
                role = Role.User,
                parts = listOf(TextPart("${args.action::class.simpleName}: ${args.payload}"))
            )
            
            val response = a2aClient.sendMessage(
                Request(MessageSendParams(message = message))
            )
            
            when (response.data) {
                is Message -> Result(true, (response.data as Message).toKoogMessage().content)
                is Task -> Result(true, (response.data as Task).status.message?.parts?.firstOrNull()?.let { 
                    (it as? TextPart)?.text 
                })
                else -> Result(false, null, "Unexpected response type")
            }
        } catch (e: Exception) {
            Result(false, null, e.message)
        }
    }
}
```

class MobileA2AServer(
    private val port: Int = 8080

    fun start() {

            protocolVersion = "0.3.0",
            preferredTransport = TransportProtocol.JSONRPC,
            capabilities = AgentCapabilities(
                streaming = true,
                pushNotifications = true
            ),
            skills = listOf(
                AgentSkill(
                    id = "mobile_operations",
                    name = "Mobile Operations",
                    description = "Execute operations on mobile device",
                    tags = listOf("mobile", "accessibility", "automation")
                )
            )
        )
        
        val server = A2AServer(
            agentExecutor = MobileAgentExecutor(),
            agentCard = agentCard
        )
        
        val transport = HttpJSONRPCServerTransport(server)
        transport.start(
            engineFactory = CIO,
            port = port,
            path = "/mobile-agent",
            wait = true
        )
    }
}

// PC端通过A2A调用Mobile
// Mobile端作为Server接收请求
class MobileAgentExecutor : AgentExecutor {
    override suspend fun execute(request: Request<MessageSendParams>): Response<*> {
        val message = request.params.message
        val content = (message.parts.first() as TextPart).text
        
        // 解析并执行Mobile操作
        val result = when {
            content.startsWith("OPEN_APP") -> openApp(content.removePrefix("OPEN_APP: "))
            content.startsWith("CLICK") -> clickElement(content.removePrefix("CLICK: "))
            else -> "Unknown command"
        }
        
        return Response(
            data = Message(
                messageId = Uuid.random().toString(),
                role = Role.Agent,
                parts = listOf(TextPart(result))
            )
        )
    }
}

val collaborativeStrategy = strategy<String, String>("collaborative") {
    val nodeAnalyze by node<String, TaskPlan> { input ->
        val pcCapabilities = withA2AAgentClient("pc-openclaw") {
            getAgentCard("pc-openclaw").capabilities
        }
        planTask(input, pcCapabilities)
    }
    
    val nodeMobileExecute by node<SubTask, String> { subTask ->
        executeOnMobile(subTask)
    }
    
    val nodePCExecute by node<SubTask, String> { subTask ->
        withA2AAgentClient("pc-openclaw") {
            sendMessageToAgent("pc-openclaw", subTask.description)
            awaitResponse()
        }
    }
    
    val nodeSync by node<List<String>, String> { results ->
        withA2AAgentClient("pc-openclaw") {
            sendMessageToAgent("pc-openclaw", "SYNC_RESULTS: ${results.joinToString()}")
            awaitResponse()
        }
    }
    
    // 构建协作流程图
    edge(nodeStart forwardTo nodeAnalyze)
    edge(nodeAnalyze forwardTo nodeMobileExecute onCondition { it.assignedTo == "mobile" })
    edge(nodeAnalyze forwardTo nodePCExecute onCondition { it.assignedTo == "pc" })
    edge(nodeMobileExecute forwardTo nodeSync)
    edge(nodePCExecute forwardTo nodeSync)
    edge(nodeSync forwardTo nodeFinish)
}
